#include <Wire.h>
#include <LiquidCrystal_I2C.h>

const int LCD_COLS = 16;
const int LCD_ROWS = 2;
const int MAX_CHARS = LCD_COLS * LCD_ROWS;

LiquidCrystal_I2C lcd(0x27, LCD_COLS, LCD_ROWS);

const String phrase = "privet nu che kak dela";
const int phraseLen = phrase.length();

int charIndex = 0;
bool completed = false;

void setup() {
  lcd.init();
  lcd.backlight();
  lcd.clear();
  lcd.setCursor(0, 0);
}

void loop() {
  if (completed || phraseLen == 0) {
    return;
  }

  if (charIndex >= phraseLen || charIndex >= MAX_CHARS) {
    completed = true;
    return;
  }

  int row = (charIndex < LCD_COLS) ? 0 : 1;
  int col = charIndex % LCD_COLS;

  lcd.setCursor(col, row);
  lcd.print(phrase[charIndex]);

  charIndex++;

  delay(1000);
}
