#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>

#define DHTPIN 2
#define DHTTYPE DHT11

LiquidCrystal_I2C lcd(0x27, 16, 2);
DHT dht(DHTPIN, DHTTYPE);

void setup() {
  lcd.begin(16, 2);
  lcd.backlight();
  dht.begin();
}

void loop() {
  lcd.clear();

  float t = dht.readTemperature();
  float h = dht.readHumidity();

  lcd.setCursor(0, 0);
  if (isnan(t)) {
    lcd.print("T: Error   ");
  } else {
    lcd.print("T: ");
    lcd.print(t, 1);
    lcd.print(" C");
  }

  lcd.setCursor(0, 1);
  if (isnan(h)) {
    lcd.print("H: Error   ");
  } else {
    lcd.print("H: ");
    lcd.print(h, 1);
    lcd.print(" %");
  }

  delay(2000);
}
